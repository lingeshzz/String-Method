class Solution {
    public String convert(String s, int numRows) {

        if (numRows == 1 || numRows >= s.length()) {
            return s;
        }

        String ans = "";

        for (int row = 0; row < numRows; row++) {

            int i = row;
            int step = 2 * (numRows - 1);

            while (i < s.length()) {

                ans = ans + s.charAt(i);

                if (row != 0 && row != numRows - 1) {
                    int j = i + step - 2 * row;

                    if (j < s.length()) {
                        ans = ans + s.charAt(j);
                    }
                }

                i = i + step;
            }
        }

        return ans;
    }
}
